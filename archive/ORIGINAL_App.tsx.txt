export default function App() {
  const [currentIndex, setCurrentIndex] = useState(0);
  const current = SANCTUARIES[currentIndex];

  // Interactive States
  const [searchFocused, setSearchFocused] = useState(false);
  const [searchQuery, setSearchQuery] = useState('');
  const [dynamicFormOpen, setDynamicFormOpen] = useState(false);
  const [nightsSlider, setNightsSlider] = useState(7);
  const [selectedTimelineIndex, setSelectedTimelineIndex] = useState(0);
  const [detailModalOpen, setDetailModalOpen] = useState(false);
  const [isAudioPlaying, setIsAudioPlaying] = useState(false);
  const [bookingConfirmed, setBookingConfirmed] = useState(false);
  const [bookingPace, setBookingPace] = useState<'Expedition' | 'Slow Living' | 'Photographic' | 'Culinary'>('Slow Living');
  const [guestCount, setGuestCount] = useState(2);
  const [activeExpIndex, setActiveExpIndex] = useState(1);

  // Animation Refs
  const organicMaskRef = useRef<HTMLDivElement>(null);
  const heroImageRef = useRef<HTMLImageElement>(null);
  const topTrajectoryPathRef = useRef<SVGPathElement>(null);
  const rockTrajectoryPathRef = useRef<SVGPathElement>(null);
  const topFollowerRef = useRef<SVGGElement>(null);
  const rockFollowerRef = useRef<SVGGElement>(null);
  const customCursorDotRef = useRef<HTMLDivElement>(null);
  const customCursorRingRef = useRef<HTMLDivElement>(null);
  const audioContextRef = useRef<AudioContext | null>(null);

  // Web Audio Synthesizer: 432Hz Resonant Drone & Ambient Waves
  const toggleAudio = () => {
    if (!isAudioPlaying) {
      try {
        const AudioContextClass = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
        const ctx = new AudioContextClass();
        audioContextRef.current = ctx;

        const osc1 = ctx.createOscillator();
        const osc2 = ctx.createOscillator();
        const gain = ctx.createGain();

        osc1.type = 'sine';
        osc1.frequency.setValueAtTime(216, ctx.currentTime);
        osc2.type = 'sine';
        osc2.frequency.setValueAtTime(432, ctx.currentTime);

        gain.gain.setValueAtTime(0.04, ctx.currentTime);

        osc1.connect(gain);
        osc2.connect(gain);
        gain.connect(ctx.destination);
        osc1.start();
        osc2.start();

        setIsAudioPlaying(true);
      } catch (e) {
        console.warn('Audio synthesis fallback', e);
        setIsAudioPlaying(true);
      }
    } else {
      if (audioContextRef.current) {
        audioContextRef.current.close();
        audioContextRef.current = null;
      }
      setIsAudioPlaying(false);
    }
  };

  // Dual-Ring Custom Trailing Cursor with Anime.js inertia
  useEffect(() => {
    const dot = customCursorDotRef.current;
    const ring = customCursorRingRef.current;
    if (!dot || !ring) return;

    let mouseX = window.innerWidth / 2;
    let mouseY = window.innerHeight / 2;
    let ringX = mouseX;
    let ringY = mouseY;
    let isMoving = true;

    const onMouseMove = (e: MouseEvent) => {
      mouseX = e.clientX;
      mouseY = e.clientY;
      dot.style.left = `${mouseX}px`;
      dot.style.top = `${mouseY}px`;
    };

    const renderCursor = () => {
      if (!ring) return;
      ringX += (mouseX - ringX) * 0.18;
      ringY += (mouseY - ringY) * 0.18;
      ring.style.left = `${ringX}px`;
      ring.style.top = `${ringY}px`;
      if (isMoving) requestAnimationFrame(renderCursor);
    };

    window.addEventListener('mousemove', onMouseMove);
    requestAnimationFrame(renderCursor);

    const hoverables = document.querySelectorAll('button, a, input, [data-interactive="true"]');
    const handleMouseEnter = () => ring.classList.add('active');
    const handleMouseLeave = () => ring.classList.remove('active');

    hoverables.forEach(el => {
      el.addEventListener('mouseenter', handleMouseEnter);
      el.addEventListener('mouseleave', handleMouseLeave);
    });

    return () => {
      isMoving = false;
      window.removeEventListener('mousemove', onMouseMove);
      hoverables.forEach(el => {
        el.removeEventListener('mouseenter', handleMouseEnter);
        el.removeEventListener('mouseleave', handleMouseLeave);
      });
    };
  }, [currentIndex, dynamicFormOpen]);

  // Master Orchestration: 20+ Anime.js Animations
  const runMasterAnimations = () => {
    if (typeof anime === 'undefined') return;

    // 1. Entrance Stagger for Navigation
    anime({
      targets: '.anime-nav-entry',
      opacity: [0, 1],
      translateY: [-16, 0],
      delay: anime.stagger(90, { start: 80 }),
      easing: 'easeOutCubic',
      duration: 800
    });

    // 3. EDITORIAL TEXT & ELEMENT REVEAL
    anime({
      targets: '.anime-char',
      opacity: [0, 1],
      translateY: [40, 0],
      skewY: [6, 0],
      delay: anime.stagger(80, { start: 200 }),
      easing: 'spring(1, 80, 12, 0)',
      duration: 1200
    });

    // Description & CTAs: fade and lift concurrently with staggered delay
    anime({
      targets: '.anime-sub-word',
      opacity: [0, 1],
      translateY: [16, 0],
      delay: anime.stagger(35, { start: 700 }),
      easing: 'easeOutQuart',
      duration: 800
    });

    anime({
      targets: '#begin-journey-container',
      opacity: [0, 1],
      translateY: [24, 0],
      easing: 'easeOutBack',
      duration: 900,
      delay: 1000
    });

    anime({
      targets: '#search-capsule-wrapper',
      opacity: [0, 1],
      translateY: [28, 0],
      easing: 'easeOutCubic',
      duration: 1000,
      delay: 1100
    });

    // 1. THE CURVED TRAJECTORY DRAWING (Path Reveal):
    // easeInOutQuart (3200ms)
    if (topTrajectoryPathRef.current) {
      anime({
        targets: topTrajectoryPathRef.current,
        strokeDashoffset: [anime.setDashoffset(topTrajectoryPathRef.current), 0],
        easing: 'easeInOutQuart',
        duration: 3200,
        delay: 200
      });
    }

    if (rockTrajectoryPathRef.current) {
      anime({
        targets: rockTrajectoryPathRef.current,
        strokeDashoffset: [anime.setDashoffset(rockTrajectoryPathRef.current), 0],
        easing: 'easeInOutQuart',
        duration: 3200,
        delay: 600
      });
    }

    // Dynamic Marker: tiny, luminous glowing "path-follower" dot
    if (topTrajectoryPathRef.current && topFollowerRef.current) {
      const pathTop = anime.path(topTrajectoryPathRef.current);
      anime({
        targets: topFollowerRef.current,
        translateX: pathTop('x'),
        translateY: pathTop('y'),
        rotate: pathTop('angle'),
        easing: 'easeInOutQuart',
        duration: 5200,
        loop: true,
        delay: 400
      });
    }

    // Secondary Path Follower to Sigiriya rock
    if (rockTrajectoryPathRef.current && rockFollowerRef.current) {
      const pathRock = anime.path(rockTrajectoryPathRef.current);
      anime({
        targets: rockFollowerRef.current,
        translateX: pathRock('x'),
        translateY: pathRock('y'),
        rotate: pathRock('angle'),
        easing: 'easeInOutSine',
        duration: 5800,
        loop: true,
        delay: 800
      });
    }

    // 2. ASYMMETRICAL ORBITAL MORPH (Interactive Image Mask):
    if (organicMaskRef.current) {
      anime({
        targets: organicMaskRef.current,
        borderRadius: [
          '40% 60% 70% 30% / 40% 50% 60% 50%',
          '60% 40% 30% 70% / 50% 60% 40% 50%',
          '45% 55% 65% 35% / 45% 45% 55% 55%',
          '40% 60% 70% 30% / 40% 50% 60% 50%'
        ],
        easing: 'easeInOutSine',
        duration: 14000,
        loop: true,
        direction: 'alternate'
      });
    }

    // Overlaying Indicator: Subtle data coordinate badges drifting vertically
    anime({
      targets: '.anime-floating-badge',
      translateY: [-6, 6],
      direction: 'alternate',
      loop: true,
      easing: 'easeInOutSine',
      duration: 3600,
      delay: anime.stagger(300)
    });

    // 20. Ambient Atmosphere Breathing (radial volcanic glow)
    anime({
      targets: '.anime-radial-glow',
      scale: [1, 1.2],
      opacity: [0.3, 0.6],
      direction: 'alternate',
      loop: true,
      easing: 'easeInOutSine',
      duration: 7000
    });

    // 22. Audio Waveform Equalizer
    anime({
      targets: '.anime-bar',
      height: ['3px', '18px'],
      direction: 'alternate',
      loop: true,
      easing: 'easeInOutQuad',
      duration: () => anime.random(350, 800),
      delay: anime.stagger(90)
    });

    // Glowing Star/Orb Breathing
    anime({
      targets: '.anime-glow-star',
      scale: [1, 1.35],
      opacity: [0.7, 1],
      direction: 'alternate',
      loop: true,
      easing: 'easeInOutSine',
      duration: () => anime.random(2000, 3500),
      delay: anime.stagger(300)
    });
  };

  useEffect(() => {
    runMasterAnimations();
  }, [currentIndex]);

  // 2. Interactive Image Mask Mouse Parallax
  const handleCanvasMouseMove = (e: React.MouseEvent<HTMLDivElement>) => {
    const rect = e.currentTarget.getBoundingClientRect();
    const x = (e.clientX - rect.left) / rect.width - 0.5;
    const y = (e.clientY - rect.top) / rect.height - 0.5;

    if (heroImageRef.current && typeof anime !== 'undefined') {
      anime({
        targets: heroImageRef.current,
        translateX: -x * 32,
        translateY: -y * 32,
        rotateX: y * 6,
        rotateY: -x * 6,
        scale: 1.05,
        easing: 'easeOutQuad',
        duration: 600
      });
    }

    if (typeof anime !== 'undefined') {
      anime({
        targets: '.anime-star-layer',
        translateX: -x * 45,
        translateY: -y * 45,
        easing: 'easeOutQuad',
        duration: 600
      });
    }
  };

  const handleCanvasMouseLeave = () => {
    if (heroImageRef.current && typeof anime !== 'undefined') {
      anime({
        targets: heroImageRef.current,
        translateX: 0,
        translateY: 0,
        rotateX: 0,
        rotateY: 0,
        scale: 1,
        easing: 'easeOutCubic',
        duration: 800
      });
    }
    if (typeof anime !== 'undefined') {
      anime({
        targets: '.anime-star-layer',
        translateX: 0,
        translateY: 0,
        easing: 'easeOutCubic',
        duration: 800
      });
    }
  };

  // Search capsule hover interaction
  const handleSearchHoverEnter = () => {
    if (typeof anime !== 'undefined') {
      anime({
        targets: '#search-capsule-wrapper',
        scale: 1.04,
        boxShadow: '0 0 35px rgba(217, 119, 70, 0.5), 0 0 15px rgba(217, 119, 70, 0.3)',
        easing: 'easeOutBack',
        duration: 400
      });
      anime({
        targets: '.anime-search-icon',
        rotate: 45,
        scale: 1.15,
        easing: 'easeOutQuad',
        duration: 350
      });
    }
  };

  const handleSearchHoverLeave = () => {
    if (!searchFocused && typeof anime !== 'undefined') {
      anime({
        targets: '#search-capsule-wrapper',
        scale: 1,
        boxShadow: '0 0 0px rgba(217, 119, 70, 0)',
        easing: 'easeOutQuad',
        duration: 400
      });
      anime({
        targets: '.anime-search-icon',
        rotate: 0,
        scale: 1,
        easing: 'easeOutQuad',
        duration: 350
      });
    }
  };

  // 15. 3D Perspective Card Tilt
  const handle3DCardMove = (e: React.MouseEvent<HTMLDivElement>) => {
    const card = e.currentTarget;
    const rect = card.getBoundingClientRect();
    const x = e.clientX - rect.left - rect.width / 2;
    const y = e.clientY - rect.top - rect.height / 2;
    const rotateX = (-y / (rect.height / 2)) * 12;
    const rotateY = (x / (rect.width / 2)) * 12;

    if (typeof anime !== 'undefined') {
      anime({
        targets: card,
        rotateX,
        rotateY,
        scale: 1.03,
        easing: 'easeOutQuad',
        duration: 350
      });
      const img = card.querySelector('.anime-card-zoom');
      if (img) {
        anime({
          targets: img,
          scale: 1.12,
          translateX: x / 25,
          translateY: y / 25,
          easing: 'easeOutQuad',
          duration: 350
        });
      }
    }
  };

  const handle3DCardLeave = (e: React.MouseEvent<HTMLDivElement>) => {
    const card = e.currentTarget;
    if (typeof anime !== 'undefined') {
      anime({
        targets: card,
        rotateX: 0,
        rotateY: 0,
        scale: 1,
        easing: 'easeOutElastic(1, 0.6)',
        duration: 700
      });
      const img = card.querySelector('.anime-card-zoom');
      if (img) {
        anime({
          targets: img,
          scale: 1,
          translateX: 0,
          translateY: 0,
          easing: 'easeOutCubic',
          duration: 700
        });
      }
    }
  };

  // 17. Magnetic Button Snap
  const handleMagneticMove = (e: React.MouseEvent<HTMLElement>) => {
    const btn = e.currentTarget;
    const rect = btn.getBoundingClientRect();
    const btnCenterX = rect.left + rect.width / 2;
    const btnCenterY = rect.top + rect.height / 2;
    const deltaX = (e.clientX - btnCenterX) * 0.4;
    const deltaY = (e.clientY - btnCenterY) * 0.4;

    if (typeof anime !== 'undefined') {
      anime({
        targets: btn,
        translateX: deltaX,
        translateY: deltaY,
        easing: 'easeOutQuad',
        duration: 250
      });
    }
  };

  const handleMagneticLeave = (e: React.MouseEvent<HTMLElement>) => {
    const btn = e.currentTarget;
    if (typeof anime !== 'undefined') {
      anime({
        targets: btn,
        translateX: 0,
        translateY: 0,
        easing: 'easeOutElastic(1, 0.6)',
        duration: 700
      });
    }
  };

  // Navigation Prev/Next
  const handleNext = () => {
    const nextIdx = (currentIndex + 1) % SANCTUARIES.length;
    transitionTo(nextIdx, 1);
  };

  const handlePrev = () => {
    const prevIdx = (currentIndex - 1 + SANCTUARIES.length) % SANCTUARIES.length;
    transitionTo(prevIdx, -1);
  };

  const transitionTo = (index: number, dir: number) => {
    if (typeof anime !== 'undefined') {
      anime({
        targets: heroImageRef.current,
        opacity: [0.4, 1],
        scale: [1.1, 1],
        translateX: [dir * 30, 0],
        easing: 'easeOutQuart',
        duration: 800
      });

      anime({
        targets: '.anime-progress-line-fill',
        width: `${((index + 1) / SANCTUARIES.length) * 100}%`,
        easing: 'easeOutElastic(1, 0.7)',
        duration: 750
      });
    }
    setCurrentIndex(index);
  };

  // Search input interaction
  const handleSearchFocus = () => {
    setSearchFocused(true);
    setDynamicFormOpen(true);
    if (typeof anime !== 'undefined') {
      anime({
        targets: '#search-capsule-wrapper',
        scale: 1.04,
        boxShadow: '0 0 35px rgba(217, 119, 70, 0.45)',
        easing: 'easeOutBack',
        duration: 450
      });
    }
  };

  // 10. Elastic Card Snap on Timeline Steps
  const handleSelectTimeline = (idx: number) => {
    setSelectedTimelineIndex(idx);
    if (typeof anime !== 'undefined') {
      anime({
        targets: '.anime-timeline-item',
        scale: (_el: any, i: number) => (i === idx ? [0.93, 1] : 0.95),
        opacity: (_el: any, i: number) => (i === idx ? 1 : 0.6),
        translateY: (_el: any, i: number) => (i === idx ? [12, 0] : 0),
        easing: 'easeOutElastic(1, 0.6)',
        duration: 900
      });
    }
  };

  // Scroll smoothly to section
  const scrollTo = (id: string) => {
    const el = document.getElementById(id);
    if (el) {
      el.scrollIntoView({ behavior: 'smooth' });
    }
  };

  return (
    <div
      onMouseMove={handleCanvasMouseMove}
      onMouseLeave={handleCanvasMouseLeave}
      className="relative min-h-screen w-full bg-[#121417] text-[#e2ded7] font-tech selection:bg-[#d97746]/30 overflow-x-hidden"
    >
      {/* 18. Custom Dual-Ring Trailing Cursor */}
